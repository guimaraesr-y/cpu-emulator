cmake_minimum_required(VERSION 3.16)
project(cpu_emulator LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BUILD_VISUALIZER "Build the native 3D laboratory" OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} ${CMAKE_BINARY_DIR}/bin)
endforeach()
add_library(cpu_core cpu/cpuFunctions.c)
add_library(lab_playback visualizer/playback.c)
target_link_libraries(lab_playback PUBLIC cpu_core)
add_library(assembler_core assembler/asm.c)
add_executable(assembler assembler.c)
target_link_libraries(assembler PRIVATE assembler_core)
add_library(lab_app visualizer/app.c)
target_link_libraries(lab_app PUBLIC lab_playback assembler_core)
add_executable(test_app tests/test_app.c)
target_link_libraries(test_app PRIVATE lab_app)
add_executable(emulator main.c)
target_link_libraries(emulator PRIVATE cpu_core)
add_executable(test_playback tests/test_playback.c)
target_link_libraries(test_playback PRIVATE lab_playback)
# Keep assert-based checks enabled in Release too.
if(MSVC)
    target_compile_options(test_playback PRIVATE /UNDEBUG)
    target_compile_options(test_app PRIVATE /UNDEBUG)
else()
    target_compile_options(test_playback PRIVATE -UNDEBUG)
    target_compile_options(test_app PRIVATE -UNDEBUG)
    foreach(target cpu_core lab_playback assembler_core lab_app test_app assembler emulator test_playback)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endif()
enable_testing()
add_test(NAME playback COMMAND test_playback)
add_test(NAME app COMMAND test_app)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME isa_end_to_end COMMAND ${CMAKE_COMMAND} -E env
        CPU_ASSEMBLER=$<TARGET_FILE:assembler> CPU_EMULATOR=$<TARGET_FILE:emulator>
        ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -v)
endif()
if(BUILD_VISUALIZER)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git
        GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
    FetchContent_MakeAvailable(raylib)
    FetchContent_Declare(raygui GIT_REPOSITORY https://github.com/raysan5/raygui.git
        GIT_TAG 25c8c65a6e5f0f4d4b564a0343861898c6f2778b)
    FetchContent_MakeAvailable(raygui)
    add_executable(cpu-lab visualizer/main.c visualizer/scene.c visualizer/editor.c visualizer/ui.c)
    target_include_directories(cpu-lab PRIVATE ${raygui_SOURCE_DIR}/src)
    target_link_libraries(cpu-lab PRIVATE lab_app raylib)
    if(NOT MSVC)
        target_compile_options(cpu-lab PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/bin/demo.bin
        COMMAND assembler ${CMAKE_SOURCE_DIR}/example_programs/program1.asm ${CMAKE_BINARY_DIR}/bin/demo.bin
        DEPENDS assembler ${CMAKE_SOURCE_DIR}/example_programs/program1.asm)
    add_custom_target(lab_demo ALL DEPENDS ${CMAKE_BINARY_DIR}/bin/demo.bin)
    add_dependencies(cpu-lab lab_demo)
endif()
